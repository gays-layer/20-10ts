        const canvasBg = document.getElementById('canvas-bg');
        const ctxBg = canvasBg.getContext('2d');
        let width, height;
        let petals = [];

        function resizeBgCanvas() {
            width = canvasBg.width = window.innerWidth;
            height = canvasBg.height = window.innerHeight;
        }
        window.addEventListener('resize', resizeBgCanvas);
        resizeBgCanvas();

        class Petal {
            constructor() {
                this.x = Math.random() * width;
                this.y = Math.random() * height - height;
                this.size = Math.random() * 12 + 8;
                this.speedY = Math.random() * 1.5 + 0.8;
                this.speedX = Math.random() * 1 - 0.5;
                this.angle = Math.random() * 360;
                this.spin = Math.random() * 2 - 1;
                this.color = ['#ffb3c1', '#ff4d6d', '#ff758f', '#ff8fa3'][Math.floor(Math.random() * 4)];
                this.opacity = Math.random() * 0.7 + 0.3;
            }
            update() {
                this.y += this.speedY;
                this.x += Math.sin(this.y / 30) + this.speedX;
                this.angle += this.spin;
                if (this.y > height) {
                    this.y = -20;
                    this.x = Math.random() * width;
                }
            }
            draw() {
                ctxBg.save();
                ctxBg.translate(this.x, this.y);
                ctxBg.rotate((this.angle * Math.PI) / 180);
                ctxBg.globalAlpha = this.opacity;
                ctxBg.fillStyle = this.color;
                ctxBg.beginPath();
                ctxBg.ellipse(0, 0, this.size, this.size / 2, 0, 0, Math.PI * 2);
                ctxBg.fill();
                ctxBg.restore();
            }
        }

        for (let i = 0; i < 50; i++) {
            petals.push(new Petal());
        }

        function animateBg() {
            ctxBg.clearRect(0, 0, width, height);
            petals.forEach(petal => {
                petal.update();
                petal.draw();
            });
            requestAnimationFrame(animateBg);
        }
        animateBg();

        const canvasFw = document.getElementById('canvas-fireworks');
        const ctxFw = canvasFw.getContext('2d');
        let fwParticles = [];

        function resizeFwCanvas() {
            canvasFw.width = window.innerWidth;
            canvasFw.height = window.innerHeight;
        }
        window.addEventListener('resize', resizeFwCanvas);
        resizeFwCanvas();

        class HeartParticle {
            constructor(x, y, color) {
                this.x = x;
                this.y = y;
                this.color = color;
                this.alpha = 1;
                this.size = Math.random() * 3 + 2;
                
                // Thuật toán tọa độ trái tim
                const t = Math.random() * Math.PI * 2;
                const speed = Math.random() * 4 + 2;
                this.vx = 16 * Math.pow(Math.sin(t), 3) * (speed / 10);
                this.vy = -(13 * Math.cos(t) - 5 * Math.cos(2*t) - 2 * Math.cos(3*t) - Math.cos(4*t)) * (speed / 10);
            }
            update() {
                this.x += this.vx;
                this.y += this.vy;
                this.alpha -= 0.015;
            }
            draw() {
                ctxFw.save();
                ctxFw.globalAlpha = this.alpha;
                ctxFw.fillStyle = this.color;
                ctxFw.beginPath();
                ctxFw.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctxFw.fill();
                ctxFw.restore();
            }
        }

        function triggerFireworks() {
            const colors = ['#e11d48', '#ff758f', '#ffb3c1', '#ffd700', '#c026d3'];
            const centerX = canvasFw.width / 2;
            const centerY = canvasFw.height / 3;

            for (let i = 0; i < 120; i++) {
                const color = colors[Math.floor(Math.random() * colors.length)];
                fwParticles.push(new HeartParticle(centerX, centerY, color));
            }
        }

        function animateFireworks() {
            ctxFw.clearRect(0, 0, canvasFw.width, canvasFw.height);
            fwParticles.forEach((p, index) => {
                p.update();
                p.draw();
                if (p.alpha <= 0) fwParticles.splice(index, 1);
            });
            requestAnimationFrame(animateFireworks);
        }
        animateFireworks();

        /* -------------------------------------------------------------
           6. ÂM NHẠC NỀN DU DƯƠNG (WEB AUDIO API)
        ------------------------------------------------------------- */
        let isPlaying = false;
        let audioCtx, oscillator, gainNode;

        function playRomanticMelody() {
            if (isPlaying) return;
            
            const AudioContext = window.AudioContext || window.webkitAudioContext;
            audioCtx = new AudioContext();
            
            // Giai điệu nhẹ nhàng (tần số nốt nhạc)
            const notes = [261.63, 329.63, 392.00, 523.25, 440.00, 349.23, 329.63, 293.66];
            let noteIndex = 0;

            function playNextNote() {
                if (!isPlaying) return;
                oscillator = audioCtx.createOscillator();
                gainNode = audioCtx.createGain();

                oscillator.type = 'sine';
                oscillator.frequency.setValueAtTime(notes[noteIndex], audioCtx.currentTime);
                
                gainNode.gain.setValueAtTime(0.01, audioCtx.currentTime);
                gainNode.gain.exponentialRampToValueAtTime(0.1, audioCtx.currentTime + 0.1);
                gainNode.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 1.2);

                oscillator.connect(gainNode);
                gainNode.connect(audioCtx.destination);

                oscillator.start();
                oscillator.stop(audioCtx.currentTime + 1.2);

                noteIndex = (noteIndex + 1) % notes.length;
                setTimeout(playNextNote, 800);
            }

            isPlaying = true;
            document.getElementById('music-icon').className = 'fas fa-volume-up text-xl text-rose-600 animate-bounce';
            playNextNote();
        }

        function stopMelody() {
            isPlaying = false;
            if (audioCtx) audioCtx.close();
            document.getElementById('music-icon').className = 'fas fa-music text-xl text-pink-600';
        }

        document.getElementById('music-toggle').addEventListener('click', () => {
            if (isPlaying) {
                stopMelody();
            } else {
                playRomanticMelody();
            }
        });
